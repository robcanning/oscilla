Cheatsheet

Quick reference for Oscilla cue syntax.

Timing & Navigation

stop()
pause()
pause(toggle)
speed(1.5)
speed(0.5,2)         # speed, duration
stopwatch()
stopwatch(hidden)
metronome(120)
metronome(120,4)     # bpm, beats

Page & Navigation

page(2)
page(next)
page(prev)
nav(loop)
nav(bounce)
nav(stop)

Interaction

button(start)
button(stop)
button(page,3)
propagate(groupId)
reuse(cueId)

Animation

scale(2)
scale(0.5,1.5)       # x, y
scale(2,3s)          # scale, duration
rotate(360)
rotate(360,5s)       # degrees, duration
fade(0)
fade(1,2s)           # opacity, duration
color(#ff0000)
color(#ff0000,3s)

Traversal

o2p(pathId)
o2p(pathId,10s)
traverse(point1,point2)
traverse(pathId)

Text

text(Hello)
text(Line1|Line2|Line3)
text(content,2s)     # text, interval

Media

audio(file.mp3)
audio(file.mp3,0.8)  # file, volume
video(file.mp4)
video(file.mp4,loop)
synth(440)
synth(440,sine)      # freq, waveform
synth(C4,triangle)

OSC

osc(/address,value)
osc(/note,64,100)
oscCtrl(fader1,/volume)
oscCtrl(button1,/trigger)

UI

ui(show)
ui(hide)
ui(toggle)
fade(0)              # hide element
fade(1,0.5s)         # fade in
